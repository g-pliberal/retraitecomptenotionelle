"""Les femmes âgées de la population : le stock de 2020 par tranche d'âge et par sexe.

    python pilote_femmes_agees.py <processus> <sortie.json> [reference]

La population, construite comme ``scripts/mesurer_surcote.py`` la construit
(``grille_large.grille("population", ...)``, ses constantes telles que le dépôt
les pose), simulée une fois (``grille_large.en_tetes``, allégée), puis pesée en
2020 comme ``grille_large.cohortes_de_l_annee`` la pèse : chaque cohorte, son
cas type au poids de 2020, fois l'effectif INSEE de son âge, fois sa survie
propre alignée à nombre de décès égal (``cout._survies_alignees``), fois ce que
son année de naissance donne à son poids (``GrilleEnTetes.part_de``). Chaque
cohorte compte aussi sans sa survie propre (« brute »), pour séparer ce que
font les poids de ce que fait la survie. Les têtes et les pensions sont
sommées par sexe, drapeau de décote (``Variante.decote``), drapeau d'inapte,
groupe, année de naissance et âge en 2020 ; la sortie les écrit en JSON, et
imprime les parts par tranche d'âge de l'EIR."""
import json, subprocess, sys, time
from pathlib import Path

RACINE = Path.cwd()
sys.path.insert(0, str(RACINE / "src"))
sys.path.insert(0, str(RACINE / "scripts"))
import grille_large as G  # noqa: E402
from retraite_notionnelle import cout as C  # noqa: E402
from retraite_notionnelle.config import Parametres  # noqa: E402
from retraite_notionnelle.simulateur import Simulateur  # noqa: E402

ANNEE = 2020
TRANCHES = ((0, 49, "age_moins_50"), (50, 54, "age_50_55"), (55, 59, "age_55_60"),
            (60, 64, "age_60_65"), (65, 69, "age_65_70"), (70, 74, "age_70_75"),
            (75, 79, "age_75_80"), (80, 84, "age_80_85"), (85, 200, "age_85_plus"))


def tranche(age: int) -> str:
    for debut, fin, nom in TRANCHES:
        if debut <= age <= fin:
            return nom
    raise ValueError(age)


def main() -> None:
    t0 = time.time()
    processus, sortie = int(sys.argv[1]), Path(sys.argv[2])
    git = lambda *a: subprocess.run(["git", *a], capture_output=True, text=True).stdout.strip()
    print("commit", git("rev-parse", "HEAD"), "arbre", git("rev-parse", "HEAD^{tree}"),
          "modifié" if git("status", "--porcelain", "--untracked-files=no") else "propre")
    p = Parametres()
    racine = p.racine_donnees
    _, valeurs = G.centiles(racine)
    variantes = (G.reference() if sys.argv[3:] == ["reference"] else
                 G.grille("population", Simulateur(p).effectifs, G.eir(racine), G.multiplicateurs(valeurs)))
    par_code = {v.cas.code: v for v in variantes}
    print(f"population : {len(variantes)} variantes ; {time.time() - t0:.0f} s", flush=True)
    tetes = G.en_tetes(p, variantes, processus, alleges=True)
    print(f"simulée : {len(tetes.pensionnes)} couples, échecs {dict(tetes.echecs)} ; {time.time() - t0:.0f} s",
          flush=True)
    poids = tetes.poids(ANNEE)
    alignees = C._survies_alignees(tetes.pensionnes, poids, ANNEE)
    vers_l_annee = tetes.macro.coefficient_prix(tetes.annee_euros, ANNEE) / 12.0
    cles: dict = {}
    for pensionne in tetes.pensionnes:
        part = poids.get(pensionne.code, 0.0)
        if part <= 0.0:
            continue
        v = par_code[pensionne.code]
        for decalage in range(-C._DEMI_TRANCHE, C._DEMI_TRANCHE + 1):
            liquidation = pensionne.liquidation(decalage)
            if ANNEE < liquidation:
                continue
            naissance = pensionne.generation + decalage
            age = ANNEE - naissance
            brute = (part * tetes.population.effectif(age, ANNEE) * tetes.part_de(pensionne.code, naissance))
            if brute <= 0.0:
                continue
            alignee = brute * alignees.get((pensionne.code, pensionne.generation, decalage), 1.0)
            montant = pensionne.pensions["actuel"] * vers_l_annee * C.coefficient_servi(
                pensionne.parts_regimes, tetes.revalorisation, liquidation, ANNEE)
            cle = (v.cas.sexe, bool(v.decote), bool(v.inapte), v.groupe, naissance, age,
                   liquidation - naissance)
            somme = cles.setdefault(cle, [0.0, 0.0, 0.0])
            somme[0] += alignee
            somme[1] += brute
            somme[2] += alignee * montant
    lignes = [list(cle) + somme for cle, somme in sorted(cles.items())]
    sortie.write_text(json.dumps({
        "commit": git("rev-parse", "HEAD"), "arbre": git("rev-parse", "HEAD^{tree}"),
        "colonnes": ["sexe", "decote", "inapte", "groupe", "naissance", "age_2020", "age_liquidation",
                     "tetes_alignees", "tetes_brutes", "tetes_alignees_fois_pension"],
        "couples": len(tetes.pensionnes), "echecs": dict(tetes.echecs), "lignes": lignes}),
        encoding="utf-8")


    def somme(garder, colonne: int) -> float:
        return sum(l[colonne] for l in lignes if garder(l))


    print("\n== retraités de 2020, par tranche d'âge de l'EIR (têtes alignées ; brutes)")
    for debut, fin, nom in TRANCHES:
        def dans(l, nom=nom):
            return tranche(l[5]) == nom
        f_a, t_a = somme(lambda l: dans(l) and l[0] == "F", 7), somme(dans, 7)
        f_b, t_b = somme(lambda l: dans(l) and l[0] == "F", 8), somme(dans, 8)
        ligne = [f"{nom} : femmes {100 * f_a / t_a:.1f} % ; brutes {100 * f_b / t_b:.1f} %"] if t_a else [nom]
        for sexe in ("F", "H"):
            s_a = somme(lambda l, s=sexe: dans(l) and l[0] == s, 7)
            d_a = somme(lambda l, s=sexe: dans(l) and l[0] == s and l[1], 7)
            s_b = somme(lambda l, s=sexe: dans(l) and l[0] == s, 8)
            d_b = somme(lambda l, s=sexe: dans(l) and l[0] == s and l[1], 8)
            if s_a:
                ligne.append(f"{sexe} : {s_a / 1e3:.0f} milliers ({s_b / 1e3:.0f}), décote {100 * d_a / s_a:.2f} %"
                             f" ({100 * d_b / s_b:.2f}), pension {somme(lambda l, s=sexe: dans(l) and l[0] == s, 9) / s_a:.0f} €")
        print("  " + " ; ".join(ligne))
    for sexe in ("F", "H"):
        s = somme(lambda l, s=sexe: l[0] == s, 7)
        print(f"  {sexe}, tous : {s / 1e3:.0f} milliers ; décote {100 * somme(lambda l, s=sexe: l[0] == s and l[1], 7) / s:.2f} %")
    print("\n== femmes nées de 1915 à 1945 : part et décote, année par année (alignées ; brutes)")
    for naissance in range(1915, 1946):
        t_a = somme(lambda l: l[4] == naissance, 7)
        if not t_a:
            continue
        f_a = somme(lambda l: l[4] == naissance and l[0] == "F", 7)
        f_b, t_b = somme(lambda l: l[4] == naissance and l[0] == "F", 8), somme(lambda l: l[4] == naissance, 8)
        d_a = somme(lambda l: l[4] == naissance and l[0] == "F" and l[1], 7)
        d_b = somme(lambda l: l[4] == naissance and l[0] == "F" and l[1], 8)
        print(f"  {naissance} ({ANNEE - naissance} ans) : femmes {100 * f_a / t_a:.1f} % ({100 * f_b / t_b:.1f}) ;"
              f" décote des femmes {100 * d_a / f_a:.2f} % ({100 * d_b / f_b:.2f})")
    print(f"\ndurée : {time.time() - t0:.0f} s")


if __name__ == "__main__":
    main()

"""Imprime, du classeur des résidents de l'EIR, feuille « CondLiquidation », les cellules des lignes demandées, colonnes A à M.

    python -I lire_cellules.py <racine du dépôt> <classeur> <première>-<dernière> ...
"""
import hashlib, sys
from pathlib import Path
racine, classeur = Path(sys.argv[1]), Path(sys.argv[2])
sys.path.insert(0, str(racine / "scripts" / "fetch"))
import drees_caracteristiques_retraites as D
from lecture_xlsx import feuilles
contenu = classeur.read_bytes()
print("octets", len(contenu), "; SHA-256", hashlib.sha256(contenu).hexdigest())
grille = feuilles(contenu, (D.FEUILLE_CONDITIONS,))[D.FEUILLE_CONDITIONS]
base = min(c for _, c in grille)
lbase = min(l for l, _ in grille)
for plage in sys.argv[3:]:
    a, b = map(int, plage.split("-"))
    for ligne in range(a, b + 1):
        l = ligne - 1 + lbase
        cellules = [f"{chr(65 + c)}{ligne}={grille[(l, c + base)]}" for c in range(13)
                    if (l, c + base) in grille and grille[(l, c + base)] not in ("", None)]
        if cellules:
            print(" | ".join(cellules))

----- confronter.py -----
"""Le stock de 2020 de la population (sortie de pilote_femmes_agees.py) contre l'EIR 2020 des résidents.

    python -I confronter.py <population.json>

L'EIR : le classeur « Caractéristiques de tous les retraités résidant en
France (EIR 2020) », feuille « CondLiquidation », recopié ici cellule par
cellule (lire_condliquidation.py) : M3 et M4, les femmes et les hommes en
milliers ; M270 à M295, la part de chaque tranche d'âge au 31 décembre 2020
parmi chaque sexe ; E3, les femmes décotées en milliers ; E270 à E294, la part
de chaque tranche parmi elles. Imprime, tranche par tranche : la part des
femmes, la décote des femmes, l'âge des femmes, la pension moyenne des femmes,
de la population (têtes alignées, puis brutes) et de l'EIR ; puis la pension
moyenne et la décote des femmes de la population, ses femmes pesées tranche
par tranche comme celles de l'EIR."""
import json, sys

TRANCHES = ("age_moins_50", "age_50_55", "age_55_60", "age_60_65", "age_65_70",
            "age_70_75", "age_75_80", "age_80_85", "age_85_plus")
BORNES = {"age_moins_50": (0, 49), "age_50_55": (50, 54), "age_55_60": (55, 59),
          "age_60_65": (60, 64), "age_65_70": (65, 69), "age_70_75": (70, 74),
          "age_75_80": (75, 79), "age_80_85": (80, 84), "age_85_plus": (85, 200)}
EIR_EFFECTIFS = {"F": 8452.0, "H": 7196.0}              # M3, M4
EIR_PARTS = {  # M270 à M295, en %
    "F": dict(zip(TRANCHES, (0.1, 0.2, 0.8, 12.0, 22.0, 22.4, 14.1, 11.9, 16.5))),
    "H": dict(zip(TRANCHES, (0.6, 0.5, 1.3, 15.3, 23.9, 23.6, 14.1, 10.4, 10.2))),
}
EIR_DECOTEES_F = 741.0                                  # E3
EIR_PARTS_DECOTEES_F = dict(zip(TRANCHES, (0.5, 0.7, 1.4, 15.7, 30.3, 18.8, 7.4, 8.6, 16.6)))  # E270 à E294

donnees = json.load(open(sys.argv[1], encoding="utf-8"))
lignes = donnees["lignes"]
print("population simulée sur", donnees["commit"], "arbre", donnees["arbre"], ";", donnees["couples"],
      "couples ; échecs", donnees["echecs"])


def tranche(age):
    return next(n for n, (a, b) in BORNES.items() if a <= age <= b)


def somme(garder, col):
    return sum(l[col] for l in lignes if garder(l))


ALIGNEES, BRUTES, PENSIONS = 7, 8, 9
eir_tetes = {s: {t: EIR_EFFECTIFS[s] * EIR_PARTS[s][t] / 100 for t in TRANCHES} for s in "FH"}
pop = {(s, t, c): somme(lambda l, s=s, t=t: l[0] == s and tranche(l[5]) == t, c)
       for s in "FH" for t in TRANCHES for c in (ALIGNEES, BRUTES, PENSIONS)}
dec = {(t, c): somme(lambda l, t=t: l[0] == "F" and l[1] and tranche(l[5]) == t, c)
       for t in TRANCHES for c in (ALIGNEES, BRUTES)}
total = {(s, c): sum(pop[s, t, c] for t in TRANCHES) for s in "FH" for c in (ALIGNEES, BRUTES)}
print("\ntranche | femmes, % : population (brutes) ; EIR | décote des femmes, % : population (brutes) ; EIR"
      " | part de la tranche parmi les femmes, % : population ; EIR | pension des femmes, € : population")
for t in TRANCHES:
    fa, ha = pop["F", t, ALIGNEES], pop["H", t, ALIGNEES]
    fb, hb = pop["F", t, BRUTES], pop["H", t, BRUTES]
    ef, eh = eir_tetes["F"][t], eir_tetes["H"][t]
    ed = EIR_DECOTEES_F * EIR_PARTS_DECOTEES_F[t] / 100
    print(f"{t} | {100 * fa / (fa + ha):.1f} ({100 * fb / (fb + hb):.1f}) ; {100 * ef / (ef + eh):.1f}"
          f" | {100 * dec[t, ALIGNEES] / fa:.2f} ({100 * dec[t, BRUTES] / fb:.2f}) ; {100 * ed / ef:.2f}"
          f" | {100 * fa / total['F', ALIGNEES]:.1f} ; {EIR_PARTS['F'][t]:.1f}"
          f" | {pop['F', t, PENSIONS] / fa:.0f}")
for s in "FH":
    print(f"{s}, tous : population {total[s, ALIGNEES] / 1e3:.0f} milliers, EIR {EIR_EFFECTIFS[s]:.0f}")
print(f"part des femmes, tous : population {100 * total['F', ALIGNEES] / (total['F', ALIGNEES] + total['H', ALIGNEES]):.1f} %"
      f" (brutes {100 * total['F', BRUTES] / (total['F', BRUTES] + total['H', BRUTES]):.1f}) ;"
      f" EIR {100 * EIR_EFFECTIFS['F'] / (EIR_EFFECTIFS['F'] + EIR_EFFECTIFS['H']):.1f} %")
for nom, poids in (("telles quelles", {t: pop["F", t, ALIGNEES] for t in TRANCHES}),
                   ("pesées comme l'EIR", {t: pop["F", t, ALIGNEES] * EIR_PARTS["F"][t] / 100
                                            * total["F", ALIGNEES] / pop["F", t, ALIGNEES]
                                            for t in TRANCHES if pop["F", t, ALIGNEES]})):
    n = sum(poids.values())
    pension = sum(poids[t] * pop["F", t, PENSIONS] / pop["F", t, ALIGNEES] for t in poids) / n
    decote = sum(poids[t] * dec[t, ALIGNEES] / pop["F", t, ALIGNEES] for t in poids) / n
    print(f"femmes de la population, {nom} : pension moyenne {pension:.0f} € ; décote {100 * decote:.2f} %")
print("EIR : pension de droit direct, majorations comprises, des femmes (M39) 1177 € ; décote des femmes (E6) 8,8 %")
print("\nfemmes nées de 1915 à 1945, l'année de naissance : part des femmes et décote des femmes, alignées (brutes)")
for naissance in range(1915, 1946):
    t = somme(lambda l: l[4] == naissance, ALIGNEES)
    if not t:
        continue
    f = somme(lambda l: l[4] == naissance and l[0] == "F", ALIGNEES)
    fb = somme(lambda l: l[4] == naissance and l[0] == "F", BRUTES)
    tb = somme(lambda l: l[4] == naissance, BRUTES)
    d = somme(lambda l: l[4] == naissance and l[0] == "F" and l[1], ALIGNEES)
    db = somme(lambda l: l[4] == naissance and l[0] == "F" and l[1], BRUTES)
    print(f"  {naissance} ({2020 - naissance} ans) : femmes {100 * f / t:.1f} % ({100 * fb / tb:.1f}) ;"
          f" décote {100 * d / f:.2f} % ({100 * db / fb:.2f})")

"""Les femmes de 80 à 84 ans et de 85 ans et plus de la population, décotées ou non : ce que la survie propre leur laisse (têtes alignées sur têtes brutes) et leur pension moyenne.

    python -I decotees.py <population.json>
"""
import json, sys
lignes = json.load(open(sys.argv[1], encoding="utf-8"))["lignes"]
for decote in (True, False):
    for debut, fin, nom in ((80, 84, "80 à 84 ans"), (85, 200, "85 ans et plus")):
        siennes = [l for l in lignes if l[0] == "F" and l[1] == decote and debut <= l[5] <= fin]
        alignees, brutes = sum(l[7] for l in siennes), sum(l[8] for l in siennes)
        pension = sum(l[9] for l in siennes) / alignees
        print(f"femmes {'décotées' if decote else 'non décotées'}, {nom} : survie propre, alignées sur brutes,"
              f" {alignees / brutes:.3f} ; pension moyenne {pension:.0f} €")
