"""La carrière longue des femmes de la population, génération 1953 (poids de 2020).

    python pilote_cl4.py 1953 2020

La mesure de 2026-11-05-etape-14-duree-des-femmes (pilote.py) refaite telle
quelle, puis, pour chaque femme qui ne part pas en carrière longue, sa
carrière rebâtie au moteur Python à 60 ans et à l'âge qui précède l'âge
légal, et les deux conditions de D. 351-1-1 lues dessus : l'entrée précoce
(cinq trimestres validés à la fin de l'année des 16, 17 et 20 ans,
CarriereLongue._entree_precoce), la durée cotisée et réputée cotisée
(cotisés + CarriereLongue._reputes_assimiles) contre la durée requise."""
import collections, re, subprocess, sys
sys.path.insert(0, "scripts")
import grille_large as G
from retraite_notionnelle.config import Parametres
from retraite_notionnelle.simulateur import Simulateur
from retraite_notionnelle.scenarios.actuel import AgesAnnulationDecote, AgesOuverture
from retraite_notionnelle.droit.commun import ligne_cotisee

print("commit", subprocess.run(["git", "rev-parse", "HEAD"], capture_output=True, text=True).stdout.strip())
p = Parametres(); racine = p.racine_donnees
effectifs = Simulateur(p).effectifs
variantes = G.grille("population", effectifs, G.eir(racine), ())
simulateur = Simulateur(p).pour_la_projection()
moteur = simulateur.scenario_actuel
cl = moteur.carriere_longue
args = [int(a) for a in sys.argv[1:]]


def lire(cas, gen, age):
    c = cas.carriere_a(simulateur, gen, age)
    an = c.annee_liquidation
    cotises = c.trimestres_cumules(l for l in c.lignes if ligne_cotisee(moteur, c, l) and l.annee <= an)
    valides = c.trimestres_cumules(l for l in c.lignes if l.annee <= an)
    reputes = cl._reputes_assimiles(c)
    natures = collections.Counter()
    for l in c.lignes:
        if l.annee <= an and not ligne_cotisee(moteur, c, l) and l.trimestres_valides > 0:
            natures[l.type_periode] += c.trimestres_retenus(l)
    premiere = min((l.annee for l in c.lignes if l.trimestres_valides > 0), default=None)
    entree = {m: cl._entree_precoce(c, an, m, 5) for m in (16, 17, 20)}
    return dict(cotises=cotises, valides=valides, reputes=reputes, natures=natures,
                debut=None if premiere is None else premiere - c.annee_naissance,
                entree=entree)


for gen, annee in zip(args[::2], args[1::2]):
    femmes = [(v, w * v.part_de_la_generation(gen))
              for v, w in zip(variantes, G.poids_retraites(variantes, effectifs, annee))
              if v.cas.sexe == "F" and w > 0]
    lectures = [G._lectures_de_la_condition(v.cas, gen) for v, _ in femmes]
    departs = G.Departs(simulateur, [a for ages, _ in lectures for a in ages],
                        [d for _, siens in lectures for d in siens], du_jour=True)
    legal = AgesOuverture(racine).age(gen)[0]; annulation = AgesAnnulationDecote(racine).age(gen)[0]
    places, conditions = collections.Counter(), collections.Counter()
    dec = collections.Counter()
    debuts_cl, debuts = collections.Counter(), collections.Counter()
    manques = collections.Counter()
    total = 0.0
    erreurs = 0
    for v, w in femmes:
        if w <= 0.0:
            continue
        try:
            age, condition = G.condition_au_depart(simulateur, v.cas, gen, departs)
            actuel = departs.au_depart(v.cas, gen)
        except (ValueError, KeyError):
            continue
        total += w
        place = G.position_du_depart(age, legal, annulation)
        places[place] += w
        conditions[condition] += w
        groupe = f"{v.groupe} {v.tranche}"
        if condition == "carriere_longue":
            dec[("1 carrière longue", groupe)] += w
            continue
        try:
            a60 = lire(v.cas, gen, 60.0)
            avant = lire(v.cas, gen, round(legal - 1 / 12, 4))
        except Exception as e:  # noqa: BLE001
            erreurs += 1
            dec[("9 carrière non rebâtie", type(e).__name__)] += w
            continue
        requis = actuel.trimestres_requis
        debuts[(place, a60["debut"])] += w
        if not a60["entree"][20]:
            dec[("2 entrée après 20 ans", groupe)] += w
            dec[("2 entrée après 20 ans, place", place)] += w
            continue
        a_60 = a60["cotises"] + a60["reputes"]
        a_avant = avant["cotises"] + avant["reputes"]
        if a_60 >= requis:
            dec[("3 ouverte à 60 ans, non prise", groupe)] += w
            dec[("3 ouverte à 60 ans, non prise, place/condition", f"{place} {condition}")] += w
            if place == "age_legal":
                print("  non prise à l'âge légal :", v.cas.code, round(age, 3), a60["cotises"], a60["reputes"], requis, round(w, 1))
            continue
        if a_avant >= requis:
            dec[("4 ouverte avant l'âge légal, non prise", groupe)] += w
            dec[("4 ouverte avant l'âge légal, non prise, place/condition", f"{place} {condition}")] += w
            if place == "age_legal":
                print("  4, non prise à l'âge légal :", v.cas.code, round(age, 3), avant["cotises"], avant["reputes"], requis, round(w, 1))
            continue
        manque = requis - a_avant
        dec[("5 entrée avant 20 ans, durée cotisée courte", groupe)] += w
        dec[("5 entrée avant 20 ans, durée cotisée courte, place", place)] += w
        tranche = ("1 à 4" if manque <= 4 else "5 à 8" if manque <= 8 else "9 à 20"
                   if manque <= 20 else "21 et plus")
        dec[("5 manque, en trimestres, à l'âge qui précède l'âge légal", tranche)] += w
        # Ce que la carrière valide sans le coter ni le réputer : ses natures.
        assim = avant["valides"] - avant["cotises"]
        for nature, t in avant["natures"].items():
            manques[nature] += w * t
        manques["réputés"] += w * avant["reputes"]
        manques["assimilés validés"] += w * assim
        manques["poids"] += w
        chom = avant["natures"].get("chomage_indemnise", 0)
        # Sans son chômage non réputé : le chômage réputé est déjà dans reputes.
        if chom - min(chom, avant["reputes"]) >= manque:
            dec[("5 dont le manque tient au seul chômage", groupe)] += w
        if assim - avant["reputes"] >= manque:
            dec[("5 dont le manque tient aux assimilés non réputés", groupe)] += w
    print(f"\n== génération {gen}, poids de {annee} ; âge légal {legal}, d'annulation {annulation}")
    print(f"carrières non rebâties : {erreurs}")
    for cle, w in sorted(places.items()):
        print(f"  place | {cle} | {100 * w / total:.1f}")
    for cle, w in sorted(conditions.items()):
        print(f"  condition | {cle} | {100 * w / total:.1f}")
    for (nom, cle), w in sorted(dec.items()):
        if 100 * w / total >= 0.1:
            print(f"  décomposition | {nom} | {cle} | {100 * w / total:.2f}")
    for (place, debut), w in sorted(debuts.items(), key=lambda kv: (kv[0][0], kv[0][1] or 0)):
        if 100 * w / total >= 0.2:
            print(f"  âge du premier trimestre validé, hors carrière longue | {place} | {debut} | {100 * w / total:.2f}")
    if manques["poids"]:
        n = manques.pop("poids")
        for nature, t in sorted(manques.items()):
            print(f"  durée courte, trimestres moyens | {nature} | {t / n:.2f}")

"""Les femmes de 1953 du salariat privé entrées avant 20 ans qui manquent la
carrière longue : la répartition de leurs années assimilées, et une carrière
type, année par année."""
import collections, sys
sys.path.insert(0, "scripts")
import grille_large as G
from retraite_notionnelle.config import Parametres
from retraite_notionnelle.simulateur import Simulateur
from retraite_notionnelle.droit.commun import ligne_cotisee
p = Parametres(); racine = p.racine_donnees
effectifs = Simulateur(p).effectifs
variantes = G.grille("population", effectifs, G.eir(racine), ())
sim = Simulateur(p).pour_la_projection(); moteur = sim.scenario_actuel; cl = moteur.carriere_longue
gen = 1953
femmes = [(v, w * v.part_de_la_generation(gen)) for v, w in zip(variantes, G.poids_retraites(variantes, effectifs, 2020))
          if v.cas.sexe == "F" and w > 0 and v.groupe == "regime_general" and v.tranche in ("30_40", "40_plus")]
tot = sum(w for _, w in femmes)
dist = collections.Counter(); par_tranche = collections.Counter()
exemples = []
for v, w in femmes:
    c = v.cas.carriere_a(sim, gen, 61.0)
    an = c.annee_liquidation
    cot = c.trimestres_cumules(l for l in c.lignes if ligne_cotisee(moteur, c, l) and l.annee <= an)
    val = c.trimestres_cumules(l for l in c.lignes if l.annee <= an)
    assim = val - cot
    e20 = cl._entree_precoce(c, an, 20, 5)
    cle = (v.tranche, e20, "0" if assim == 0 else "1-8" if assim <= 8 else "9-16" if assim <= 16 else "17-24" if assim <= 24 else "25+")
    dist[cle] += w; par_tranche[v.tranche] += w
    exemples.append((w, v, c, cot, val))
print("femmes du salariat privé, tranches 30_40 et 40_plus, 1953, part de toutes ces femmes :")
for k, w in sorted(dist.items()):
    print(" ", k, f"{100*w/tot:.1f}")
print({k: round(100*w/tot, 1) for k, w in par_tranche.items()})
exemples.sort(key=lambda e: -e[0])
for w, v, c, cot, val in [e for e in exemples if e[1].tranche == "30_40"][:1] + [e for e in exemples if e[1].tranche == "40_plus"][:1]:
    print("\n", v.cas.code, "poids", round(w, 5), "cotisés", cot, "validés", val, "enfants", v.cas.nombre_enfants)
    for l in c.lignes:
        print(f"   {l.annee} {l.annee - c.annee_naissance} {l.type_periode} tv={l.trimestres_valides} cot={int(ligne_cotisee(moteur, c, l))} nat={l.reputes_cotises_nature}")
