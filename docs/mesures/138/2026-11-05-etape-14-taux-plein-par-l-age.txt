"""Les départs au taux plein par l'âge de la population, génération par génération.

    python pilote_age.py 1953 2020 [1955 2022 ...]

Chaque couple : une génération, simulée au portage, et l'année dont elle prend
les poids de retraités (``grille_large.poids_retraites``), multipliés par la
part que la génération donne à chaque variante (``part_de_la_generation``).
Tous les départs, sans restriction d'âge ni recalage de la page. Pour chaque
sexe : la part des départs au taux plein par l'âge (``condition_au_depart``),
hors des jumelles inaptes, et leur âge ; la part des départs à l'âge
d'annulation de la décote et après lui, et la part du taux plein par l'âge
dans chacune ; les variantes qui y partent, par groupe et tranche."""
import collections, subprocess, sys, time
sys.path.insert(0, "scripts")
import grille_large as G
from retraite_notionnelle.config import Parametres
from retraite_notionnelle.simulateur import Simulateur
from retraite_notionnelle.scenarios.actuel import AgesAnnulationDecote, AgesOuverture

t0 = time.time()
print("commit", subprocess.run(["git", "rev-parse", "HEAD"], capture_output=True, text=True).stdout.strip())
p = Parametres(); racine = p.racine_donnees
effectifs = Simulateur(p).effectifs
variantes = G.grille("population", effectifs, G.eir(racine), ())
print(f"population : {len(variantes)} variantes ; {time.time()-t0:.0f} s")
simulateur = Simulateur(p).pour_la_projection()
args = [int(a) for a in sys.argv[1:]]
for gen, annee in zip(args[::2], args[1::2]):
    poids = G.poids_retraites(variantes, effectifs, annee)
    tous = [(v, w * v.part_de_la_generation(gen)) for v, w in zip(variantes, poids) if w > 0]
    tous = [(v, w) for v, w in tous if w > 0]
    lectures = [G._lectures_de_la_condition(v.cas, gen) for v, _ in tous]
    departs = G.Departs(simulateur, [a for ages, _ in lectures for a in ages],
                        [d for _, siens in lectures for d in siens], du_jour=True)
    legal = AgesOuverture(racine).age(gen)[0]; annulation = AgesAnnulationDecote(racine).age(gen)[0]
    print(f"\n== génération {gen}, poids de {annee} ; âge légal {legal:.2f}, d'annulation {annulation:.2f}")
    for sexe in ("F", "H"):
        total = tpa = somme_ages = 0.0
        places, conditions_aux_places = collections.Counter(), collections.Counter()
        ages, ages_exacts, qui = collections.Counter(), collections.Counter(), collections.Counter()
        inaptes_tpa = 0.0
        for v, w in tous:
            if v.cas.sexe != sexe:
                continue
            try:
                age, condition = G.condition_au_depart(simulateur, v.cas, gen, departs)
            except (ValueError, KeyError):
                continue
            total += w
            place = G.position_du_depart(age, legal, annulation)
            places[place] += w
            conditions_aux_places[(place, condition + (" (inapte)" if v.inapte else ""))] += w
            if condition != "taux_plein_age":
                continue
            if v.inapte:
                inaptes_tpa += w
                continue
            tpa += w; somme_ages += w * age
            ages[G.age_du_depart(age)] += w
            ages_exacts[round(age, 2)] += w
            qui[f"{v.groupe} {v.tranche}"] += w
        print(f"  {sexe} : taux plein par l'âge, hors jumelles inaptes, {100 * tpa / total:.2f} % des départs ;"
              f" âge moyen {somme_ages / tpa if tpa else float('nan'):.2f} ; jumelles inaptes au taux plein par l'âge {100 * inaptes_tpa / total:.2f} %")
        print("    places : " + " ; ".join(f"{k} {100 * w / total:.2f}" for k, w in sorted(places.items())))
        for (place, condition), w in sorted(conditions_aux_places.items()):
            if place in ("age_annulation", "apres_annulation"):
                print(f"    à la place {place}, {condition} : {100 * w / total:.2f} % des départs, {100 * w / places[place]:.1f} % de la place")
        for nom, compte in (("âge révolu", ages), ("âge exact", ages_exacts), ("qui", qui)):
            print(f"    {nom} : " + " ; ".join(f"{k} {100 * w / tpa:.1f}" for k, w in sorted(compte.items()) if 100 * w / tpa >= 0.05))
    sys.stdout.flush()
print(f"\ndurée : {time.time()-t0:.0f} s")
